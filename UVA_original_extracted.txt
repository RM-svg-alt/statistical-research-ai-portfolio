SAS Code Sample for Biostatistics Interview

Purpose: The sample code demonstrates registry data processing, clinical cohort derivation, automated descriptive reporting, regression modeling, trend analyses, visualization, flexible modeling, and hospital-volume analyses.

Confidentiality: Dataset names, file paths, and institution-specific identifiers have been generalized. No patient-level output or protected health information is included.

1. Data Import and Cohort Creation

Imports a de-identified registry file, restricts to index procedures, flags syndrome status, maps primary procedure codes to clinically interpretable groups, derives log-transformed operative variables, and creates clinically meaningful birth-weight and neonatal categories.

PROC IMPORT OUT= WORK.Can
    DATAFILE= "[de-identified-path]\registry_puf.csv"
    DBMS=CSV REPLACE;
    GETNAMES=YES;
    DATAROW=2;
RUN;

data Can_;
    set Can;

    if IndexProc = 1;

    if max(of SY_:)=1 then syndrome_ = 1;
    else syndrome_ = 0;

    /* VSD repair */
    if primproc = 110 then procedure_desc = 'VSD';
    else if primproc = 5001 then procedure_desc = 'VSD';

    /* TOF repair */
    else if primproc in (350, 360, 370, 3330, 3340, 5004) then procedure_desc = 'TOF repair';

    /* Complete atrioventricular canal repair */
    else if primproc = 170 then procedure_desc = 'AVC (AVSD) repair';

    /* Arterial switch operation */
    else if primproc = 1110 then procedure_desc = 'Arterial switch operation';
    else if primproc = 1120 then procedure_desc = 'Arterial switch operation (ASO) and VSD repair';

    /* Glenn / Fontan excluded from final analytic cohort */
    else if primproc in (1670,1680,1690,1700,2130) then procedure_desc = 'Glenn';
    else if primproc in (970,980,1000,1010,2780,2790,3310,3320) then procedure_desc = 'Fontan';

    /* Other target procedures */
    else if primproc = 230 then procedure_desc = 'Truncus';
    else if primproc = 870 then procedure_desc = 'Norwood';
    else if primproc in (1210,1220,1230,1240,1250,1280,3460,3470) then procedure_desc = 'Coarctation repair';

    if procedure_desc in ('Glenn','Fontan') then delete;
    if procedure_desc ^= " ";

    CPBTm_     = log(CPBTm);
    XClampTm_  = log(XClampTm);
    Weightkg_  = log(Weightkg);
    BirthWtKg_ = log(BirthWtKg);

    if BirthWtKg >= 0 and BirthWtKg < 1 then weight_category = '0-1 kg';
    else if BirthWtKg >= 1 and BirthWtKg < 2 then weight_category = '1-2 kg';
    else if BirthWtKg >= 2 and BirthWtKg < 3 then weight_category = '2-3 kg';
    else if BirthWtKg >= 3 and BirthWtKg < 4 then weight_category = '3-4 kg';
    else if BirthWtKg >= 4 and BirthWtKg < 5 then weight_category = '4-5 kg';
    else weight_category = '>5 kg';

    if AgeDays <= 28 then neonate = 1;
    else neonate = 0;

    if weight_category ^= '>5 kg';

    if BirthWtKg < 2.5 then low_birth_weight = 1;
    else if BirthWtKg >= 2.5 then low_birth_weight = 0;
    else low_birth_weight = .;

    TLOS = LOS + AgeDays;
run;

2. Clinical Subsetting and Descriptive Queries

Creates procedure-specific cohorts and answers targeted clinical questions such as Norwood patients who doubled weight within 90 days and mortality in extremely low birth-weight infants.

data Can_nor;
    set Can_;
    if procedure_desc = 'Norwood';
run;

proc freq data=Can_;
    tables procedure_desc;
run;

/* Norwood: doubling of weight within <3 months (<=90 days) */
proc sql;
    select
        count(*) as norwood_total,
        sum(BirthWtKg > 0 and WeightKg > 0 and AgeDays <= 90 and WeightKg >= 2*BirthWtKg)
            as norwood_doubled_lt3mo,
        calculated norwood_doubled_lt3mo / calculated norwood_total * 100
            as pct_doubled_lt3mo format=8.2
    from Can_nor
    where procedure_desc = 'Norwood';
quit;

data norwood_flag;
    set Can_nor;
    where procedure_desc = 'Norwood';
    doubled_lt3mo = (BirthWtKg > 0 and WeightKg > 0 and AgeDays <= 90 and WeightKg >= 2*BirthWtKg);
    if doubled_lt3mo = 1;
run;

proc sql;
    select
        count(*) as N_lt1kg,
        sum(MtOpD=1) as Deaths_lt1kg,
        calculated Deaths_lt1kg / calculated N_lt1kg * 100 as MortalityPct format=8.2
    from Can_
    where BirthWtKg < 1 and BirthWtKg > 0;
quit;

3. Descriptive Statistics and Table 1 Automation

Reusable macros generate categorical and continuous descriptive summaries by clinical group, including p-values and formatted output for publication-style Table 1 reporting.

%Macro TB_Catg(Data,By);
    data &Data._TB;
        set &Data.;
        Total = 1;
        &By._ = put(&By.,20.);
    run;

    %let i = 1;
    %do %until(not %length(%scan(&Catg,&i)));
        proc freq data=&Data._TB noprint;
            tables &By._*%scan(&Catg,&i) / chisq outpct out=Catg_OutFreq;
            output out = Catg_Stats(keep = P_PCHI) chisq;
        run;

        proc freq data=&Data._TB noprint;
            tables Total*%scan(&Catg,&i) / outpct out=Catg_OutFreq_Tot;
        run;

        data Catg_Stats;
            set Catg_Stats;
            if P_PCHI < .001 then p_val_chi = '<0.001';
            else if P_PCHI < 0.05 then p_val_chi = round(P_PCHI, 0.001);
            else p_val_chi = round(P_PCHI, 0.01);
            keep p_val_chi;
        run;

        %let i = %eval(&i+1);
    %end;
%Mend;

%Macro TB_NonContP(Data, By);
    data &Data._TB;
        set &Data.;
        Total = 1;
        &By._ = put(&By.,20.);
    run;

    %let i = 1;
    %do %until(not %length(%scan(&Cont_NonP,&i)));
        proc npar1way data=&Data._TB noprint Wilcoxon;
            class &By._;
            var %scan(&Cont_NonP,&i);
            output out = ContNonP_Stats(keep = P_KW) Wilcoxon;
        run;

        proc means data=&Data._TB noprint;
            class &By._;
            var %scan(&Cont_NonP,&i);
            output out = NonP_Des N=N P25=P25 P50=P50 P75=P75;
        run;

        %let i = %eval(&i+1);
    %end;
%Mend;

%let Cont_NonP = BirthWtKg GestAgeWeeks AgeDays WeightKg Heightcm CPBTm XClampTm Readmit30 LOS PLOS;
%TB_NonContP(Can_, weight_category);

%let Catg = Gender Premature AntenatalDiag syndrome_ OpType ReOpInAdm AgeCat Emergency MTopd;
%TB_Catg(Can_, weight_category);

4. Logistic Regression and GLIMMIX Modeling

Fits multivariable mortality models using birth weight as both categorical and continuous exposure, adjusting for operative and patient-level covariates.

proc logistic data=Can_;
    where premature ^= 3;
    class Gender Premature AntenatalDiag syndrome_(ref='0') OpType
          AgeCat(ref='Infant') low_birth_weight(ref='0');
    model MTopd(ref='2') = low_birth_weight syndrome_ OpType WeightKg CPBTm_ XClampTm_ AgeCat Premature
        / selection = stepwise;
run;

proc logistic data=Can_;
    where premature ^= 3;
    class Gender Premature AntenatalDiag syndrome_(ref='0') OpType AgeCat(ref='Infant');
    model MTopd(ref='2') = BirthWtKg syndrome_ OpType WeightKg CPBTm_ XClampTm_ AgeCat Premature
        / selection = stepwise;
run;

proc glimmix data=Can_ method=quad;
    where premature ^= 3;
    class weight_category(ref='4-5 kg') Gender Premature AntenatalDiag OpType
          AgeCat(ref='Infant') Premature(ref='2') syndrome_(ref='0');
    model MTopd(ref='2') = weight_category CPBTm_ XClampTm_ AgeCat Premature syndrome_
        / dist=binary link=logit or cl;
run;

proc glimmix data=Can_ method=quad;
    where premature ^= 3;
    class Gender Premature AntenatalDiag OpType AgeCat(ref='Infant') syndrome_(ref='0');
    model MTopd(ref='2') = BirthWtKg CPBTm_ XClampTm_ AgeCat Premature syndrome_
        / dist=binary link=logit or cl;
run;

5. Trend Analyses and Procedure-Year Summaries

Evaluates temporal patterns in procedures and patient characteristics, including low birth-weight trends and annual procedure counts in selected weight strata.

data procedures;
    set Can_;
    if 1.5 <= WeightKg <= 2.0 then weight_group = '1500g to 2000g';
run;

data weight_filtered;
    set procedures;
    where weight_group = '1500g to 2000g';
run;

proc freq data=weight_filtered;
    tables SurgYear*procedure_desc / nopercent norow nocol out=procedure_counts;
run;

proc sort data=procedure_counts;
    by procedure_desc SurgYear;
run;

data procedure_increase;
    set procedure_counts;
    by procedure_desc;
    retain first_year first_count;

    if first.procedure_desc then do;
        first_year = SurgYear;
        first_count = count;
    end;

    if last.procedure_desc then do;
        last_year = SurgYear;
        last_count = count;
        increase = ((last_count - first_count) / first_count) * 100;
        output;
    end;
run;

proc print data=procedure_increase;
    var procedure_desc first_year last_year first_count last_count increase;
run;

proc sql;
    create table proportion_by_surgyear as
    select SurgYear,
           (sum(premature = 1) / count(*) * 100) as proportion_percent
    from Can_
    group by SurgYear;
quit;

6. Visualization and Odds-Ratio Plotting

Creates outcome visualizations, procedure-specific plots, and adjusted odds-ratio plots to support manuscript-style interpretation.

proc sql;
    create table work.death_percentage as
    select procedure_desc,
           Weightkg_,
           sum(case when MtOpD = 1 then 1 else 0 end) as total_deaths,
           count(*) as total_count,
           calculated total_deaths / calculated total_count * 100 as death_percentage
    from Can_
    group by procedure_desc, Weightkg_;
quit;

proc sgplot data=work.death_percentage;
    by procedure_desc;
    scatter x=Weightkg_ y=death_percentage / markerattrs=(symbol=circlefilled);
    title 'Weight vs Mortality Rate for Each Procedure';
    xaxis label='Weight (kg)';
    yaxis label='Percentage of Deaths (%)';
run;

proc sgplot data=Can_;
    vbox AgeDays / category=MTopd group=procedure_desc;
    xaxis label='Survival Status';
    yaxis label='Age at Surgery (Days)';
    title 'Age at Surgery by Survival Status';
run;

proc glimmix data=Can_ method=quad;
    where premature ^= 3;
    class Gender Premature AntenatalDiag OpType AgeCat(ref='Infant');
    model MTopd(ref='2') = BirthWtKg_ CPBTm_ XClampTm_ AgeCat Premature /
        dist=binary link=logit solution cl oddsratio;
    ods output OddsRatios=ORs_logBW;
run;

proc sgplot data=ORs_logBW;
    scatter y=Label x=Estimate / xerrorlower=Lower xerrorupper=Upper
        markerattrs=(symbol=circlefilled size=10);
    refline 1 / axis=x lineattrs=(pattern=shortdash thickness=1);
    xaxis type=log logbase=10 grid label='Adjusted Odds Ratio for Mortality';
    yaxis display=(noticks nolabel);
    title 'Adjusted Effect of Birth Weight on Mortality';
run;

7. Flexible Nonlinear Modeling

Uses GAM/spline modeling to explore nonlinear relationships between age at surgery and mortality in procedure-specific cohorts.

data FP;
    set Can_;
    where BirthWtKg > 0 and premature ^= 3;
run;

data test;
    set Can_;
    if procedure_desc = 'Coarctation repair';
    if AgeDays <= 30;
run;

proc gam data=test plots=component(clm);
    where procedure_desc = 'Coarctation repair';
    model MTopd(event='1') = spline(AgeDays) / dist=binomial;
    output out=gam_output pred=predicted;
run;

proc sgplot data=Can_;
    where procedure_desc = 'Norwood';
    loess x=AgeDays y=MTopd / smooth=0.4 clm;
    xaxis label='Age at Surgery (Days)';
    yaxis label='Probability of Mortality';
    title 'Norwood: Mortality Probability by Age at Surgery';
run;

8. Hospital Volume Analysis

Aggregates procedures at the hospital level, derives average annual volume, classifies hospitals into volume groups, and merges hospital-volume metrics back to patient-level analytic data.

proc sql;
    create table Hosp_Volume as
    select STS_HospID,
           count(*) as Total_Procedures,
           count(distinct SurgYear) as Years
    from Can_
    group by STS_HospID;
quit;

data Hosp_Volume;
    set Hosp_Volume;
    Avg_Annual_Volume = Total_Procedures / Years;

    if Avg_Annual_Volume >= 48 then VolumeGroup = 'High';
    else VolumeGroup = 'Low';
run;

proc means data=Hosp_Volume p50 p25 p75 p90 p10;
    var Avg_Annual_Volume;
run;

proc freq data=Hosp_Volume;
    tables VolumeGroup;
run;

proc sql;
    create table Final_Data as
    select a.*, b.Avg_Annual_Volume, b.VolumeGroup
    from Can_ as a
    left join Hosp_Volume as b
        on a.STS_HospID = b.STS_HospID;
quit;

proc freq data=Final_Data;
    where premature ^= 3;
    tables premature*VolumeGroup / all;
run;

proc freq data=Final_Data;
    tables low_birth_weight*VolumeGroup / all;
run;
